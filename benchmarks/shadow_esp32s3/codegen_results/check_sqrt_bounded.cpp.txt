#include <cstdint>
#include <cstdio>
#include <chrono>
uint32_t candidate(uint32_t value) {
  if (value < 4) return value == 0 ? 0 : 1;
  int bits = 32 - __builtin_clz(value);
  uint32_t root = 3u << ((bits + 1) / 2 - 2);
  root = (root + value / root) / 2;
  root = (root + value / root) / 2;
  if (root == 65536 || root * root > value) --root;
  return root;
}
int main() {
 uint32_t expected=0; uint64_t next=1;
 for (uint64_t v=0; v<(1ULL<<18); ++v) {
   if(v==next) { ++expected; next=uint64_t(expected+1)*(expected+1); }
   uint32_t result=candidate(v);
   if(result!=expected) { printf("FAIL %llu actual=%u expected=%u\n", (unsigned long long)v,result,expected); return 1; }
 }
 puts("PASS all 262144 inputs below 2^18");
}
